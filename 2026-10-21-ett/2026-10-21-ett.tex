\begin{frame}{Long data}
    \textit{Big data} means many records, typically over a short span

    \medskip

    Municipal records often run continuously for decades $\rightarrow$ \textit{long data}

    \medskip

    Long data can illustrate trends, establish precedents, enable comparisons, and uncover context

    \medskip

    Long data also poses challenges around accessibility, standards, and comparability
            
    \begin{center}
        \begin{tabular}{lll}
            \toprule
            \textbf{Holding} & \textbf{Accessible?} & \textbf{Amount} \\
            \midrule
            Digital files with text & Yes & Small \\
            Scanned pages           & No  & Large \\
            Paper, film, audio      & No  & Unknown \\
            \bottomrule
        \end{tabular}
    \end{center}

    \bottomcite{Samuel Arbesman, ``Stop Hyping Big Data and Start Paying Attention to `Long Data','' \textit{Wired}, January 29, 2013}
\end{frame}

\begin{frame}{Document understanding}
    \begin{columns}
        \begin{column}{.65\textwidth}
            \textit{Document understanding} is a research field, and a set of software frameworks, that turns documents into data

            \medskip
            
            \begin{itemize}\small
                \item \textbf{Layout analysis} finds the headings, paragraphs, tables, and figures on a page, and the order to read them in
                \item \textbf{Optical character recognition (OCR)} reads printed or handwritten text from a picture of a page
                \item \textbf{Table recognition} rebuilds a table's rows, columns, and cells
                \item \textbf{Field extraction} pulls named items like dates or totals out of forms
            \end{itemize}

            \medskip
            
            AI models are rapidly improving at converting complex PDFs into accessible formats: text, Markdown, and HTML
        \end{column}
        \begin{column}{.3\textwidth}\centering
            \includegraphics[width=\textwidth]{img/datalab.png}
            {\footnotesize From \url{https://www.datalab.to}}
        \end{column}
    \end{columns}

    \bottomcite{Yihao Ding et al., ``A Survey on MLLM-based Visually Rich Document Understanding: Methods, Challenges, and Emerging Trends,'' \textit{ACL 2026}, pp.~13319--13340}
\end{frame}

\begin{frame}{Data liberation and rescue}
    \textit{Data liberation} moves information that is already public out of the formats that bury it (scans, PDFs, portals) and into open and accessible data formats

    \medskip

    \textit{Data rescue} is a grassroots, interdisciplinary movement to preserve public datasets from decay and censorship

    \medskip

    \begin{itemize}\setlength\itemsep{3pt}
        \item \textbf{2014.} The Sunlight Foundation runs a PDF Liberation Hackathon to build shared tools for pulling tables out of government PDFs
        \item \textbf{2017 on.} The Public Utility Data Liberation project turns federal energy filings into an open and documented database
        \item \textbf{2020.} The U.S. Supreme Court rules in \textit{Georgia v. Public.Resource.Org} that the state's annotated code cannot be copyrighted
        \item \textbf{2025 on.} Researchers improvise technical capacity to archive data before it is deleted by the Trump administration
    \end{itemize}

    \bottomcite{Tom Lee, ``PDF Liberation: Why It Matters and How You Can Help,'' Sunlight Foundation, January 24, 2014; \textit{Georgia v. Public.Resource.Org, Inc.}, 590 U.S. 255 (2020)}
\end{frame}

\begin{frame}{Data liberation for the City of Boulder}
    \begin{itemize}\setlength\itemsep{4pt}
        \item \textbf{Hotline answers.} Helping staff research answers to questions from council and other departments
        \item \textbf{Budget history.} Converting PDF tables into accessible data formats
        \item \textbf{The law on a date.} Which version of an ordinance was in force when a permit was issued
        \item \textbf{Records requests.} A document made searchable once is ready for every later request
        \item \textbf{Accessibility.} Every scanned page gains text that a screen reader can speak
        \item \textbf{Research and reporting.} Residents, reporters, and researchers work from the same corpus as staff
    \end{itemize}
\end{frame}

\begin{frame}{Council hotline}\small
    Council members and staff exchange questions and answers via the council hotline

    \medskip

    797 messages since March 2023: 209 questions from council and 144 answers from staff

    \medskip

    Many of those answers point to budgets, presentations, ordinances, and reports already in the city's archives

    \bigskip

    \begin{alertblock}{Question}
        How many of council's questions could be answered using the city's own records?
    \end{alertblock}

    \bottomcite{City of Boulder open data, ``\href{https://bouldercolorado.gov/boulder-council-hotline-archives}{Emails to the City Council Hotline},'' March 2, 2023 to October 15, 2026}

\begin{frame}{Analyzing hotline: answers from staff}
    \begin{columns}
        \begin{column}{0.55\textwidth}

            \medskip

            \medskip

\begin{frame}{Accessibility and CORA/FOIA}
    \begin{columns}\small
        \begin{column}{.65\textwidth}
            The ADA requires the city's web content, including PDFs, to meet accessibility guidelines (WCAG 2.1) by April 2027

            \medskip
            
            Scanned documents without text layers and figures without alt text are endemic throughout municipal archives
        
            \medskip
        
            Documents needed to get a service must be readable; archived scans are exempt
        
            \medskip
        
            Fulfilling CORA requests can be expensive and complex $\rightarrow$ data liberation and rescue could save costs

            \medskip

            Back of the envelope: OCR for 10k budget pages at 0.75\textcent{} a page is about \$75, roughly two hours of research billed at CORA's \$41.37 hourly cap
        \end{column}
        \begin{column}{.3\textwidth}
            \includegraphics[width=.95\textwidth]{img/responsive.jpg}
        \end{column}
    \end{columns}
    \bottomcite{28 C.F.R. \S\S 35.200--35.201, as amended by 91 Fed.\ Reg.\ 20902 (April 20, 2026); C.R.S. \S 24-34-802 (HB21-1110, HB24-1454); CORA, C.R.S. \S 24-72-201 et seq.}
\end{frame}

\begin{frame}{Community values}\small
    \begin{columns}
        \begin{column}{.65\textwidth}
            The city's Sustainability, Equity and Resilience (SER) Framework commits the city to a government that:

        \begin{itemize}
            \item ``Models stewardship, resilience, and sustainability of the city's financial, human, \textbf{information}, and physical assets''
            \item ``Supports strategic decision-making with \ldots{} timely and \textbf{reliable data and analysis}''
            \item ``Enhances and facilitates \textbf{transparency}, accuracy, efficiency, effectiveness''
            \item ``Disaggregates data\ldots{} to ensure programs, policies, and financial decisions do not have a disparate impact''
        \end{itemize}
    
        \medskip
    
        Uneven digitization and accessibility limit the ability of elected officials, staff, researchers, and journalists to study disparate impacts
        \end{column}
        \begin{column}{.3\textwidth}
            \includegraphics[width=\textwidth]{img/iceberg.jpg}

\begin{frame}{What happens next}
    \begin{columns}\small 
        \begin{column}{0.65\textwidth}
            Profound improvements in performance per dollar using document understanding to support digitization, accessibility, transparency

            \medskip

            Identify materials commonly requested by council and the public

            \medskip
            
            Convert and document them in accessible formats through the existing Open Data team and infrastructure

            \medskip

            Decisions about what to digitize, how to prioritize, where to archive, and who to open tooling to are unavoidably political

            \medskip

            Protect staff time for substantive analysis and expert interpretation rather than menial information retrieval and cleaning

            \medskip

            Implement SER Framework values through open standards and products, not expensive and fragile vendor solutions
        \end{column}
        \begin{column}{0.3\textwidth}
            \centering
            \includegraphics[width=0.9\textwidth]{img/georgi.jpg}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{Why I might be wrong}
    \begin{columns}\small 
        \begin{column}{0.56\textwidth}
            Major pivot in data culture requiring procurement, training, and maintenance

            \medskip
            
            Demand for agentic interfaces into municipal archives may be low --- or introduce novel political, legal, and operational risks

            \medskip

            Early mover risks with dramatic changes in architectures, standards, and practices for AI, agents, MCP servers, and data lakes

            \medskip

            Staff may already find most answers with the search tools they have, and know where to look

            \medskip

            Many answers may not be in archives --- new data, analyses, or judgments needed in many hotline responses

            \medskip

            Models can hallucinate plausible but wrong numbers, which could cost more trust than the time it saved
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=0.9\textwidth]{img/chart.jpg}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{Other risks}\small 
    \begin{columns}
        \begin{column}{0.65\textwidth}
            Documents in archives fall into at least three categories: \textbf{already data}, \textbf{could become data}, or \textbf{shouldn't become data}

            \medskip

            Surveillance video, customer records, code complaints, and records about victims and juveniles should \textit{not} become data

            \medskip

            What gets scanned gets cited $\rightarrow$ paper archives omitted from trends and comparisons

            \medskip

            OCR does worst on old, handwritten, tabular, and non-English pages

            \medskip

            Use the SER Framework to guide digitization priorities
        \end{column}
        \begin{column}{0.3\textwidth}
            \centering
            \includegraphics[width=.75\textwidth]{img/ignoring.jpg}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Key takeaways}
    \begin{enumerate}\setlength\itemsep{4pt}
        \item The city holds long data of immense value locked away in PDFs and paper
        \item Digitizing is getting better and cheaper, while the cost of verifying records stays roughly constant
        \item The hotline analysis shows more than half of staff answers draw on three or more kinds of records
        \item Accessibility law, CORA, and the city's own framework already call for this work
    \end{enumerate}

\end{frame}
